结构体、枚举和特征定义也可以与常量字段成员一起提供，它们可用于需要在它们之间共享常量的情况。

```rust
trait Circular {
    const PI: f64 = 3.14; // 这里定义了常量，所有实现该特征的结构体都能使用
    fn area(&self) -> f64;
}

struct Circle {
    red: f64
}

impl Circular for Circle {
    fn area(&self) -> f64{
        Circle::PI * self.red * self.red
    }
}

fn fn_circle(){
    let c_one = Circle {red: 4.2 };
    let c_two = Circle { red: 75.2 };
    println!("Area of circle one:{}", c_one.area());
    println!("Area of circle two:{}", c_two.area());
}

enum Item{
    One, Two

struct Food { cc: u32 }

    println!("Item DEFAILT_COUNT: {}", Item::DEFAILT_COUNT);
```
